Guía de Despliegue y Arquitectura Industrial

Implementación monolítica de HMI, intermediario Google Apps Script y persistencia en Google Sheets.

1. Frontend HMI

Archivo único HTML/JS con gestión de cola secuencial, sincronización de intervalos y visualización Chart.js.

2. Backend Apps Script

Intermediario seguro con funciones setup(), idempotencia por ID único y llamadas a la API de OpenAI.

3. Google Sheets

Hoja centralizada Registros con columnas estructuradas para trazabilidad industrial completa.

Guía paso a paso interactiva para desplegar el backend

1
Crear Hoja de Cálculo en Google Sheets

Cree una nueva hoja de cálculo en Google Drive y asígnele el nombre exacto Registros a la primera pestaña.

2
Abrir Editor de Apps Script

En el menú superior de Google Sheets, seleccione Extensiones > Apps Script. Borre cualquier código predeterminado.

3
Pegar el código de Code.gs

Copie el código completo mostrado más abajo y péguelo en el archivo Code.gs. Guarde el proyecto con el icono de disquete.

4
Ejecutar la función setup()

Seleccione la función setup en el selector superior del editor y haga clic en Ejecutar. Esto creará los encabezados de columnas requeridos en la hoja Registros y autorizará los permisos iniciales.

5
Desplegar como Aplicación Web

Haga clic en Implementar > Nueva implementación. Seleccione el tipo Aplicación web.

  • Descripción: HMI Data Logger v1
  • Ejecutar como: Yo (su cuenta de Google)
  • Quién tiene acceso: Cualquier usuario (Anyone)
6
Copiar URL de Ejecución e Iniciar HMI

Copie la URL generada (debe terminar en /exec), péguela en la pestaña Demostración de esta HMI y desbloquee el sistema.

Code.gs (Google Apps Script Completo)

Este script gestiona la recepción idempotente por ID único, la estructura de columnas requerida (Timestamp, ID, Zona, Variable, Valor, Unidad, Observación, Origen), el historial por GET y la integración opcional con OpenAI. 

/**
 * T1 · HMI Data Logger Inteligente - Google Apps Script Backend (Code.gs)
 * Columnas requeridas en pestaña "Registros":
 * Timestamp | ID | Zona | Variable | Valor | Unidad | Observación | Origen
 */

function setup() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Registros");
  if (!sheet) {
    sheet = ss.insertSheet("Registros");
  }
  var headers = ["Timestamp", "ID", "Zona", "Variable", "Valor", "Unidad", "Observación", "Origen"];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold");
  sheet.setFrozenRows(1);
}

function doPost(e) {
  try {
    var lock = LockService.getScriptLock();
    // Esperar hasta 10 segundos para evitar escrituras concurrentes conflictivas
    if (!lock.tryLock(10000)) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "error",
        message: "Servidor ocupado. Intente nuevamente."
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Registros");
    if (!sheet) {
      setup();
      sheet = ss.getSheetByName("Registros");
    }

    var action = data.action || "log";

    // Acción de análisis con OpenAI desde el servidor
    if (action === "analyze") {
      var apiKey = data.apiKey;
      if (!apiKey) {
        lock.releaseLock();
        return ContentService.createTextOutput(JSON.stringify({
          status: "error",
          message: "API Key de OpenAI no proporcionada al servidor."
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      // Obtener últimos 30 registros reales de Sheets
      var rows = sheet.getDataRange().getValues();
      var history = rows.slice(1).slice(-30); // Últimas 30 filas
      var promptText = "Analiza los siguientes registros reales de temperatura industrial de la caldera principal (Z-01). Describe tendencias, posibles anomalías, distingue hechos de recomendaciones y no inventes mediciones:\n" + JSON.stringify(history);

      var aiResponse = callOpenAI(apiKey, promptText);
      lock.releaseLock();
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        analysis: aiResponse
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // Acción estándar de registro con Idempotencia por ID único
    var recordId = data.id;
    var rows = sheet.getDataRange().getValues();
    
    // Verificar si el ID ya existe (Idempotencia)
    if (recordId) {
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][1] === recordId) {
          lock.releaseLock();
          return ContentService.createTextOutput(JSON.stringify({
            status: "success",
            message: "Registro ya existente (idempotencia confirmada)",
            id: recordId
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }
    }

    var timestamp = data.timestamp || new Date().toISOString();
    var zona = data.zona || "Caldera Z-01";
    var variable = data.variable || "Temperatura";
    var valor = data.valor || 0;
    var unidad = data.unidad || "°C";
    var observacion = data.observacion || "Normal";
    var origen = data.origen || "HMI Automática";

    sheet.appendRow([timestamp, recordId, zona, variable, valor, unidad, observacion, origen]);
    lock.releaseLock();

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "Registro guardado correctamente",
      id: recordId
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Registros");
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        data: []
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var rows = sheet.getDataRange().getValues();
    var headers = rows[0];
    var result = [];

    for (var i = 1; i < rows.length; i++) {
      var row = rows[i];
      result.push({
        timestamp: row[0],
        id: row[1],
        zona: row[2],
        variable: row[3],
        valor: row[4],
        unidad: row[5],
        observacion: row[6],
        origen: row[7]
      });
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      data: result
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function callOpenAI(apiKey, prompt) {
  var url = "https://api.openai.com/v1/chat/completions";
  var payload = {
    model: "gpt-4o-mini",
    messages: [
      { role: "system", content: "Eres un ingeniero industrial experto en control de procesos térmicos HMI. Analiza estrictamente los datos provistos sin inventar mediciones, separando hechos confirmados de recomendaciones técnicas." },
      { role: "user", content: prompt }
    ],
    temperature: 0.3
  };

  var options = {
    method: "post",
    contentType: "application/json",
    headers: {
      "Authorization": "Bearer " + apiKey
    },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };

  try {
    var response = UrlFetchApp.fetch(url, options);
    var json = JSON.parse(response.getContentText());
    if (json.choices && json.choices.length > 0) {
      return json.choices[0].message.content;
    } else {
      return "Error en respuesta de OpenAI: " + response.getContentText();
    }
  } catch (e) {
    return "Excepción al conectar con OpenAI: " + e.toString();
  }
}
